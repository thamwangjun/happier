```typescript
/**
 * MCP tool enable/disable configuration (CLI-local settings)
 *
 * Defines the McpToolsSettingsV1 schema and reader function.
 * The schema is validated from the `mcpToolsSettingsV1` field in ~/.happier/settings.json.
 * The reader accepts an already-loaded Settings object (not a file path) so it stays pure
 * and testable without filesystem access.
 */

import * as z from 'zod';
import { logger } from '@/ui/logger';
import type { Settings } from '@/persistence';

export const McpToolsSettingsV1Schema = z.preprocess(
    (raw) => {
        if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return {};
        return raw;
    },
    z.object({
        v: z.literal(1).default(1 as const),
        tools: z.record(z.string(), z.object({ enabled: z.boolean() })).default({}),
    }),
);

export type McpToolsSettingsV1 = z.infer<typeof McpToolsSettingsV1Schema>;

export const DEFAULT_MCP_TOOLS_SETTINGS: McpToolsSettingsV1 = { v: 1, tools: {} };

/**
 * Reads and validates the mcpToolsSettingsV1 field from a Settings object.
 *
 * Always returns a valid McpToolsSettingsV1 — never throws, never returns null.
 * - Absent key → returns DEFAULT_MCP_TOOLS_SETTINGS (silent, per D-05).
 * - Parse failure → emits logger.warn and returns DEFAULT_MCP_TOOLS_SETTINGS.
 * - Valid payload → returns parsed value.
 */
export function readMcpToolsSettingsV1(settings: Settings): McpToolsSettingsV1 {
    const raw = settings.mcpToolsSettingsV1;
    if (raw === undefined || raw === null) {
        return DEFAULT_MCP_TOOLS_SETTINGS;
    }
    const parsed = McpToolsSettingsV1Schema.safeParse(raw);
    if (!parsed.success) {
        logger.warn(`[mcpToolsSettings] mcpToolsSettingsV1 failed schema validation — using defaults. Error: ${parsed.error.message}`);
        return DEFAULT_MCP_TOOLS_SETTINGS;
    }
    return parsed.data;
}
```

Key decisions:
- `import * as z from 'zod'` — matches persistence.ts import style (per PATTERNS.md)
- `z.preprocess` wrapping — absorbs null/non-object raw values before z.object sees them (per Pattern 1 in RESEARCH.md)
- `z.literal(1).default(1 as const)` — allows blobs with no `v` field to parse (pitfall 2 prevention). If TypeScript rejects `1 as const` in this Zod version, use `.default(1)` instead (open question A2 in RESEARCH.md)
- `.default({})` on tools — allows `{ "v": 1 }` with no tools key to parse cleanly (SCHEMA-02)
- No `.strict()` — forward-compatible with future schema additions
- `export type McpToolsSettingsV1` — explicit named type alias, not left to z.infer at use sites
- Absent-key fast path before safeParse — no logger.warn on absent key (D-05)
- `import type { Settings }` — type-only import prevents circular runtime dependency with persistence.ts

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| disk → memory | `~/.happier/settings.json` is user-writable plaintext; any value can appear in `mcpToolsSettingsV1` |
| memory → module | `readMcpToolsSettingsV1` receives a `Settings` object already loaded by `readSettings()`; the field value is `unknown` at runtime |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Tampering | `mcpToolsSettingsV1` field in settings.json | mitigate | Zod schema validation via `safeParse` before any value is used; malformed values are rejected and replaced with `DEFAULT_MCP_TOOLS_SETTINGS` |
| T-01-02 | Denial of Service | Reader throwing on bad input | mitigate | `safeParse` (not `parse`) is mandatory — reader never throws; parse failures return the safe default |
| T-01-03 | Information Disclosure | `logger.warn` logging raw parse error detail | accept | The error message from Zod is schema-structural only (field names, types); it contains no user secrets or plaintext session data. Low-value disclosure. |
| T-01-04 | Tampering | `v` field set to a future version number | mitigate | `z.literal(1)` rejects any `v` other than `1`; reader falls back to default and warns |
| T-01-05 | Elevation of Privilege | Tool name injection via crafted settings key | accept | Tool names are used only as map keys for a lookup — no eval, no dynamic require, no shell execution in this module. Phase 2 filtering is a read-only predicate. |
</threat_model>
